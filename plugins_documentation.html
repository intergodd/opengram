Обзор архитектуры плагинов

Opengram поддерживает гибридную систему плагинов, позволяющую использовать как высокопроизводительные нативные C++ динамические библиотеки (.dll / .so), так и простые скриптовые плагины на Python, совместимые с экосистемой ExteraGram.

Поддерживаемые типы файлов:

• JavaScript плагины: размещаются в папке plugins/js/ с расширением .js.
• C++ плагины: размещаются в папке plugins/cxx/ в виде скомпилированных динамических библиотек.
• Python плагины: размещаются в папке plugins/installed/ с расширением .plugin или .plg.

JavaScript Скриптовые Плагины (QJSEngine)

JavaScript плагины работают во встроенной «песочнице» Qt внутри главного процесса, не требуя компиляции или тяжелых фоновых приложений. Они идеально подходят для быстрого создания функций, перехвата команд и кастомных кнопок меню.

Регистрация плагина

Каждый JS-плагин представляет собой файл с расширением .js в папке plugins/js/. Он должен вызывать opengram.registerPlugin() и передавать объект с настройками и обработчиками событий.

opengram.registerPlugin({
    name: "JS Модератор",
    version: "1.0.0",
    description: "Простой плагин для удаления участников на JavaScript",
    author: "Raitorinkus",

    // Вызывается при загрузке плагина
    onLoad: function() {
        opengram.log("JS Модератор успешно загружен!");
        // Добавляем кнопку в контекстное меню чата
        opengram.addMenuItem("js_kick_deleted", "Удалить удалённые аккаунты (JS)", "msg_user_remove");
    },

    // Вызывается при выгрузке плагина
    onUnload: function() {
        opengram.log("JS Модератор выгружен!");
    },

    // Хук исходящих сообщений (возврат false отменяет отправку)
    onSendMessage: function(chatId, text, replyToId, replyPath) {
        if (text === ".js_help") {
            opengram.sendMessage(chatId, "Доступные команды: .js_help", replyToId);
            return false; // отменить отправку оригинального сообщения .js_help в чат
        }
        return true; // продолжить обычную отправку сообщения
    },

    // Вызывается при клике на зарегистрированную кнопку в меню чата
    onMenuItemClick: function(itemId, chatId) {
        if (itemId === "js_kick_deleted") {
            opengram.log("Начато удаление удалённых аккаунтов в чате " + chatId);
            
            // Вызов нативного MTProto запроса
            opengram.sendRequest("channels_getParticipants", { chat_id: chatId, offset: 0, limit: 200 }, function(response, error) {
                if (error) {
                    opengram.log("Ошибка получения участников: " + error.text);
                    return;
                }
                
                var users = response.users;
                var deletedCount = 0;
                
                for (var i = 0; i < users.length; i++) {
                    var user = users[i];
                    if (user.deleted) {
                        deletedCount++;
                        opengram.log("Удаляем пользователя: " + user.first_name + " (ID: " + user.id + ")");
                        // Вызов бана
                        opengram.sendRequest("channels_editBanned", { chat_id: chatId, user_id: user.id }, function(res, err) {
                            if (err) opengram.log("Ошибка при удалении: " + err.text);
                        });
                    }
                }
                opengram.log("Всего найдено и удаляется удалённых аккаунтов: " + deletedCount);
            });
        }
    }
});

Доступные функции глобального API opengram (или api)

Метод Описание
log(text) Записывает строку логов в консоль отладки клиента.
addMenuItem(id, text, icon) Добавляет кнопку в меню чата (три точки вверху или клик на чат). icon может быть msg_user_remove или msg_autodelete_1m_solar.
removeMenuItem(id) Удаляет добавленную кнопку из меню.
sendMessage(chatId, text, replyToId) Отправляет сообщение в чат с указанным ID. replyToId опционален.
sendRequest(type, params, callback) Выполняет асинхронный MTProto запрос к API Telegram. Поддерживает channels_getParticipants, channels_editBanned, messages_deleteChatUser.

Нативные C++ Плагины

C++ плагины загружаются динамически во время запуска клиента с использованием механизма QLibrary. Они имеют полный доступ к C++ структурам данных сессии, чатам, пользователям и сообщениям, обеспечивая максимальное быстродействие.

Интерфейс CxxPlugin

Каждый C++ плагин должен наследоваться от базового класса Plugins::CxxPlugin и реализовывать его виртуальные функции:

Telegram/SourceFiles/plugins/cxx_plugin.h
#pragma once

#include <QtCore/QString>
#include <vector>

class PeerData;
struct MsgId;

namespace Plugins {

struct CxxMenuItem {
    QString id;
    QString text;
    QString icon;
};

class CxxPlugin {
public:
    virtual ~CxxPlugin() = default;

    // Метаданные плагина
    virtual QString name() const = 0;
    virtual QString version() const = 0;
    virtual QString description() const = 0;
    virtual QString author() const = 0;

    // Жизненный цикл
    virtual void onLoad() {}
    virtual void onUnload() {}

    // Хук исходящих сообщений (возврат false отменяет отправку)
    virtual bool onSendMessage(
        not_null<PeerData*> peer,
        QString &text,
        MsgId replyToId,
        const QString &replyPath) {
        return true;
    }

    // Регистрация кастомных кнопок в меню (3 точки)
    virtual std::vector<CxxMenuItem> menuItems() {
        return {};
    }

    // Обработка нажатия на кнопку в меню
    virtual void onMenuItemClick(
        const QString &itemId,
        not_null<PeerData*> peer) {}
};

} // namespace Plugins

Пример простого C++ плагина (Модерация / User Manager)

Ниже представлен исходный код плагина модерации, который отслеживает отправку команд и добавляет кнопки в меню чата.

#include "cxx_plugin.h"
#include <QtCore/QMessageBox>

class UserManagerCxx : public Plugins::CxxPlugin {
public:
    QString name() const override { return "C++ User Manager"; }
    QString version() const override { return "1.0.0"; }
    QString description() const override { return "Нативный плагин модерации чатов"; }
    QString author() const override { return "Developer"; }

    void onLoad() override {
        // Инициализация
    }

    bool onSendMessage(
        not_null<PeerData*> peer,
        QString &text,
        MsgId replyToId,
        const QString &replyPath) override {
        
        if (text == ".cxx_test") {
            text = "Привет из нативного C++ плагина!";
        }
        return true;
    }

    std::vector<Plugins::CxxMenuItem> menuItems() override {
        return {
            { "cxx_ban_deleted", "Удалить удаленные аккаунты (C++)", "msg_user_remove" }
        };
    }

    void onMenuItemClick(
        const QString &itemId,
        not_null<PeerData*> peer) override {
        if (itemId == "cxx_ban_deleted") {
            // Реализация удаления пользователей
        }
    }
};

// Экспортируемая функция фабрики
extern "C" {
    __declspec(dllexport) Plugins::CxxPlugin* create_plugin() {
        return new UserManagerCxx();
    }
}

Сборка C++ плагина

Скомпилируйте данный код как динамическую библиотеку (DLL на Windows) с использованием MSVC:

cl /LD /EHsc /I"путь_к_Qt/include" /I"путь_к_opengram/Telegram/SourceFiles/plugins" user_manager_cxx.cpp /link /OUT:user_manager.dll

После сборки поместите user_manager.dll в директорию plugins/cxx/ рядом с исполняемым файлом Opengram.

Интерцептор исходящих команд:

Любое исходящее сообщение, начинающееся с точки (например, .bandel, .debugmode), перехватывается плагинами. Если плагин обрабатывает эту команду, он возвращает HookStrategy.CANCEL, предотвращая отправку текста команды в сеть.

Консоль отладки и Логи

Для разработчиков плагинов в Opengram встроена консоль отладки реального времени. Чтобы ее открыть, отправьте в любой чат команду:

.debugmode

или её алиас:

.devmode

Откроется окно консоли, в которое в реальном времени будут выводиться все системные сообщения, вызовы self.log() из Python-плагинов, ошибки интерпретатора и сообщения логов C++ плагинов.
